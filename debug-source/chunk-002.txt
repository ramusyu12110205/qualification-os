function localDate(d=new Date()){
  const y=d.getFullYear(), m=String(d.getMonth()+1).padStart(2,'0'), day=String(d.getDate()).padStart(2,'0');

}

    }));
    const {data:ps,error:pe}=await sb.from('problems').upsert(rows,{onConflict:'user_id,subject_id,name'}).select('id,name');
    if(pe){alert('問題保存に失敗しました: '+pe.message);return}
    const links=(ps||[]).map(p=>({user_id:currentUser.id,session_id:session.id,problem_id:p.id,role:'new',result:null}));
    if(links.length){const {error:le}=await sb.from('study_session_problems').insert(links);if(le){alert('問題リンクの保存に失敗しました: '+le.message);return}}

}

function calcStreak(){
  const days=[...new Set(sessions.map(s=>s.study_date))].sort().reverse(); if(!days.length||days[0]!==today())return 0;
  let n=0;for(let i=0;i<days.length;i++){if(days[i]===addDays(today(),-i))n++;else break}return n;
}
