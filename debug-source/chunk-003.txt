function hideInlinePanels(except){
  ['qualification','record','past','import'].forEach(x=>{
    if(x===except)return;
    const p=document.getElementById('panel-'+x),a=document.getElementById('action-'+x);
    if(p)p.style.display='none';
    if(a)a.classList.remove('active');
  });
}
function toggleInlinePanel(type){
  const panel=document.getElementById('panel-'+type),action=document.getElementById('action-'+type);
  if(!panel||!action)return;
  const open=panel.style.display!=='none';
  hideInlinePanels(open?null:type);
  if(open){
    panel.style.display='none'; action.classList.remove('active'); return;
  }
  panel.style.display='block'; action.classList.add('active');
  if(type==='qualification') renderQualificationMasterInline();
  if(type==='record') renderRecordManagerInline();
  if(type==='past') renderPastStudyInline();
  if(type==='import') renderImportInline();
  setTimeout(()=>panel.scrollIntoView({behavior:'smooth',block:'nearest'}),0);
}

async function addQualification(){
  const name=document.getElementById('newQName')?.value.trim();
  const exam_date=document.getElementById('newQDate')?.value||null;
  const raw=document.getElementById('newQSubjects')?.value||'';
  if(!name){alert('資格名を入力してください');return}
  if(!currentUser?.id){alert('ログイン状態を確認してください。');return}

  let q=null;

  // 既存（アーカイブ済みを含む）があれば再利用・復元。
  const {data:existing,error:findError}=await sb.from('qualifications')
    .select('*')
    .eq('user_id',currentUser.id)
    .eq('name',name)
    .maybeSingle();

  if(findError){
    alert('資格の確認に失敗しました: '+findError.message);
    return;
  }

  if(existing){
    const {data:updated,error:updateError}=await sb.from('qualifications')
      .update({exam_date,archived:false})
      .eq('id',existing.id)
      .eq('user_id',currentUser.id)
      .select().single();
    if(updateError){
      alert('資格の更新に失敗しました: '+updateError.message);
      return;
    }
    q=updated;
  }else{
    const {data:created,error:createError}=await sb.from('qualifications')
      .insert({user_id:currentUser.id,name,exam_date})
      .select().single();
    if(createError){
      alert('資格の登録に失敗しました: '+createError.message);
      return;
    }
    q=created;
  }

  const names=[...new Set(raw.split(/\r?\n|,|、/).map(x=>x.trim()).filter(Boolean))];

  if(names.length){
    // 同名科目が既にあっても再利用できるように1件ずつ処理。
    for(let i=0;i<names.length;i++){
      const subjectName=names[i];
      const {data:oldSubject,error:subjectFindError}=await sb.from('subjects')
        .select('*')
        .eq('user_id',currentUser.id)
        .eq('qualification_id',q.id)
        .eq('name',subjectName)
        .maybeSingle();

      if(subjectFindError){
        alert('科目の確認に失敗しました: '+subjectFindError.message);
        return;
      }

      if(oldSubject){
        const {error:subjectUpdateError}=await sb.from('subjects')
          .update({archived:false,sort_order:i})
          .eq('id',oldSubject.id)
          .eq('user_id',currentUser.id);
        if(subjectUpdateError){
          alert('科目の更新に失敗しました: '+subjectUpdateError.message);
          return;
        }
      }else{
        const {error:subjectInsertError}=await sb.from('subjects')
          .insert({user_id:currentUser.id,qualification_id:q.id,name:subjectName,sort_order:i});
        if(subjectInsertError){
          alert('科目の登録に失敗しました: '+subjectInsertError.message);
          return;
        }
      }
    }
  }

  await loadAll();
  renderQualificationMasterInline();
  toast('資格を登録しました');
}

async function updateQualification(id){
  const name=document.getElementById('qn-'+id)?.value.trim();
  const exam_date=document.getElementById('qd-'+id)?.value||null;
  const raw=document.getElementById('qs-'+id)?.value||'';
  if(!name){alert('資格名を入力してください');return}

  const {error}=await sb.from('qualifications')
    .update({name,exam_date})
    .eq('id',id).eq('user_id',currentUser.id);
  if(error){alert('資格の更新に失敗しました: '+error.message);return}

  const old=subjects.filter(s=>s.qualification_id===id);
  const names=[...new Set(raw.split(/\r?\n|,|、/).map(x=>x.trim()).filter(Boolean))];
  const existingNames=old.map(s=>s.name);

  const add=names.filter(n=>!existingNames.includes(n));
  if(add.length){
    const {error:e}=await sb.from('subjects').insert(
      add.map((n,i)=>({user_id:currentUser.id,qualification_id:id,name:n,sort_order:old.length+i}))
    );
    if(e){alert('科目の追加に失敗しました: '+e.message);return}
  }

  const remove=old.filter(s=>!names.includes(s.name));
  if(remove.length){
    const {error:e}=await sb.from('subjects')
      .update({archived:true})
      .in('id',remove.map(s=>s.id))
      .eq('user_id',currentUser.id);
    if(e){alert('科目の更新に失敗しました: '+e.message);return}
  }

  await loadAll();
  renderQualificationMasterInline();
  toast('保存しました');
}

async function archiveQualification(id){
  if(!confirm('この資格を削除しますか？\n関連する科目・問題も参照できなくなります。'))return;
  const {error}=await sb.from('qualifications')
    .update({archived:true})
    .eq('id',id).eq('user_id',currentUser.id);
  if(error){alert('資格の削除に失敗しました: '+error.message);return}
  await loadAll();
  renderQualificationMasterInline();
  toast('資格を削除しました');
}
